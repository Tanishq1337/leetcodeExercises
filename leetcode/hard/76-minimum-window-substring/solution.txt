class Solution {
public:
    bool isEqual(const vector<int>& a, const vector<int>& b){
        for(int i = 0; i < 256; i++){
            if(a[i] < b[i])
                return false;
        }

        return true;
    }
    string minWindow(string s, string t) {
        if(s.size() < t.size()) return "";
        
        vector<int> need(256, 0);
        vector<int> have(256, 0);

        for(auto x : t)
            need[x]++;
        
        int n = s.size();
        int left = 0;
        int res = INT_MAX;
        int startIdx = -1;

        for(int right = 0; right < n; right++){
            have[s[right]]++;
            int len = right - left + 1;

            while(isEqual(have, need)){
                if(len < res){
                    startIdx = left;
                    res = len;
                }
                have[s[left]]--;
                left++;
                len = right - left + 1;
            }
